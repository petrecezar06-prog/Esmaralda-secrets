// Antetele CORS necesare pentru a permite apelurile din GitHub Pages const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization', }; export default { async fetch(request, env, ctx) { // 1. Tratarea cererii PREFLIGHT (OPTIONS) if (request.method === 'OPTIONS') { return new Response(null, { status: 204, headers: corsHeaders }); } // 2. Procesarea cererilor POST de la frontend if (request.method === 'POST') { try { const body = await request.json(); const userMessage = body.message || ''; // Dacă ai legat Workers AI (Llama model) în Cloudflare Dashboard sub variabila AI: let aiResponseText = ''; if (env.AI) { const aiResponse = await env.AI.run('@cf/meta/llama-3.1-8b-instruct', { messages: [ { role: 'system', content: 'Ești Esmeralda, o asistentă virtuală prietenoasă, inteligentă și utilă.' }, { role: 'user', content: userMessage } ] }); aiResponseText = aiResponse.response; } else { aiResponseText = `Sunt Esmeralda. Am primit mesajul tău: "${userMessage}", dar binding-ul AI nu este configurat încă în Worker.`; } // Returnăm răspunsul JSON însoțit de antetele CORS return new Response(JSON.stringify({ reply: aiResponseText }), { status: 200, headers: { 'Content-Type': 'application/json', ...corsHeaders } }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { 'Content-Type': 'application/json', ...corsHeaders } }); } } // Răspuns default pentru alte tipuri de cereri (ex: GET) return new Response(JSON.stringify({ message: 'Worker-ul Esmeralda este activ.' }), { status: 200, headers: { 'Content-Type': 'application/json', ...corsHeaders } }); } };
